#To find the twin prime numbers
#
def Prime(n):
    List=[]
    L=[]
    for j in range(2,n):
        for i in range(2,j):
            if j%i==0:
                break
        else:
            L.append(j)
    for k in L:
        for t in L:
            if k==(t-2):
                List.append((k,t))
    
    print("The co primes are:",List)
n=1000
Prime(n)

    Done=0
